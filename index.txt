    <script>
        firebase.initializeApp(firebaseConfig);
        const auth = firebase.auth();
        const db = firebase.firestore();

        const AppState = {
            currentUser: null,
            currentRole: null,
            activeTab: 'orders'
        };

        auth.onAuthStateChanged(async (user) => {
            if (user) {
                try {
                    const doc = await db.collection('employees').doc(user.uid).get();
                    if (doc.exists && doc.data().isActive) {
                        AppState.currentUser = doc.data();
                        AppState.currentRole = doc.data().role;
                        initInterface();
                    } else {
                        alert('Доступ заблокирован. Вас нет в базе данных сотрудников.');
                        auth.signOut();
                    }
                } catch (e) {
                    alert('Ошибка проверки прав доступа.');
                    auth.signOut();
                }
            } else {
                document.getElementById('app-container').classList.add('hidden');
                document.getElementById('auth-screen').classList.remove('hidden');
}
});
document.getElementById('auth-form').addEventListener('submit', (e) => {
e.preventDefault();
const email = document.getElementById('auth-email').value;
const password = document.getElementById('auth-password').value;
auth.signInWithEmailAndPassword(email, password).catch(err => alert('Ошибка: ' + err.message));
});
document.getElementById('btn-logout').addEventListener('click', () => auth.signOut());
document.querySelectorAll('.nav-item').forEach(btn => {
btn.addEventListener('click', (e) => {
document.querySelectorAll('.nav-item').forEach(b => b.classList.remove('active'));
e.currentTarget.classList.add('active');
AppState.activeTab = e.currentTarget.dataset.target;
switchTab(AppState.activeTab);
});
});
function initInterface() {
document.getElementById('auth-screen').classList.add('hidden');
document.getElementById('app-container').classList.remove('hidden');
document.getElementById('user-display-name').textContent = AppState.currentUser.name;
document.getElementById('user-display-role').textContent = AppState.currentRole;
if (AppState.currentRole !== 'admin') document.getElementById('nav-admin').classList.add('hidden');
if (AppState.currentRole === 'courier') document.getElementById('nav-inventory').classList.add('hidden');
switchTab(AppState.activeTab);
}
function switchTab(tabName) {
const container = document.getElementById('main-content');
container.innerHTML = '';
if (tabName === 'orders') {
container.innerHTML = <div class="tab-header"> <h2>Лента заказов</h2> </div> <div id="orders-feed">Загрузка данных в реальном времени...</div>;
db.collection('orders').orderBy('timestamps.created', 'desc').onSnapshot(snap => {
const feed = document.getElementById('orders-feed');
if (!feed) return;
feed.innerHTML = snap.empty ? 'Нет активных заказов' : '';
snap.forEach(doc => {
const order = doc.data();
const item = document.createElement('div');
item.className = order-card status-${order.status};
item.innerHTML = <div class="card-row"> <span class="order-number">Заказ №${order.orderNumber}</span> <span class="order-price">${order.financials.total} ₽</span> </div> <div class="card-row" style="color: var(--text-muted); font-size: 0.9rem;"> <span>Покупатель: ${order.customer.name}</span> </div> <div class="card-row" style="margin-top: 10px; font-size: 0.85rem;"> <span>📍 ${order.customer.address}</span> </div>;
feed.appendChild(item);
});
});
} else if (tabName === 'inventory') {
container.innerHTML = <h2>Компоненты и склад</h2><p style="color: var(--text-muted); margin-top:15px;">Модуль управления остатками.</p>;
} else if (tabName === 'admin') {
container.innerHTML = <h2>Управление персоналом</h2><p style="color: var(--text-muted); margin-top:15px;">Панель создания сотрудников.</p>;
}
}
